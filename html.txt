<script>
const catalog = [
 {name:"Eaglercraft",category:"Aventura",emoji:"🧱",color:"#245c48",desc:"Explore um mundo de blocos.",url:"https://eaglercraft.com/"},
 {name:"Subway Surfers",category:"Corrida",emoji:"🏃",color:"#b34d25",desc:"Corra, desvie e pegue moedas.",url:"https://poki.com/en/g/subway-surfers"},
 {name:"Temple Run 2",category:"Corrida",emoji:"🏃‍♂️",color:"#77502d",desc:"Fuja e desvie dos obstáculos.",url:"https://poki.com/"},
 {name:"Moto X3M",category:"Corrida",emoji:"🏍️",color:"#146c91",desc:"Supere pistas cheias de desafios.",url:"https://poki.com/"},
 {name:"2048",category:"Arcade",emoji:"🔢",color:"#66533d",desc:"Combine números e alcance 2048.",url:"https://play2048.co/"},
 {name:"Krunker",category:"Ação",emoji:"🎯",color:"#6650a0",desc:"Ação em um jogo de tiro online.",url:"https://krunker.io/"},
 {name:"Cookie Clicker",category:"Arcade",emoji:"🍪",color:"#92512b",desc:"Clique para criar uma fábrica de biscoitos.",url:"https://orteil.dashnet.org/cookieclicker/"},
 {name:"Jogos de aventura",category:"Aventura",emoji:"🗺️",color:"#286a61",desc:"Descubra mais mundos e desafios.",url:"https://poki.com/"}
];

let category = "Todos";
let favorites = [];
try { favorites = JSON.parse(localStorage.getItem("gz-favorites") || "[]"); } catch {}

const container = document.getElementById("games");
const search = document.getElementById("search");
const empty = document.getElementById("empty");
const title = document.getElementById("section-title");

function render(){
  const term = search.value.trim().toLowerCase();
  const list = catalog.filter(g =>
    (category === "Todos" || (category === "Favoritos"
      ? favorites.includes(g.name) : g.category === category)) &&
    (g.name.toLowerCase().includes(term) ||
     g.category.toLowerCase().includes(term) ||
     g.desc.toLowerCase().includes(term))
  );

  title.textContent = category === "Favoritos"
    ? "⭐ Seus favoritos" : "🎮 " + category;

  container.innerHTML = list.map(g => `
    <article class="game">
      <div class="cover" style="background:${g.color}">${g.emoji}</div>
      <div class="game-info">
        <h3>${g.name}</h3>
        <p>${g.desc}</p>
        <div class="actions">
          <a class="play" href="${g.url}" target="_blank"
             rel="noopener noreferrer">JOGAR ↗</a>
          <button class="fav" data-fav="${g.name}"
            aria-label="Favoritar ${g.name}"
            aria-pressed="${favorites.includes(g.name)}">
            ${favorites.includes(g.name) ? "⭐" : "☆"}
          </button>
        </div>
      </div>
    </article>
  `).join("");

  empty.hidden = list.length !== 0;
}

container.addEventListener("click", e => {
  const button = e.target.closest("[data-fav]");
  if (!button) return;
  const name = button.dataset.fav;
  favorites = favorites.includes(name)
    ? favorites.filter(n => n !== name)
    : [...favorites, name];
  try { localStorage.setItem("gz-favorites", JSON.stringify(favorites)); } catch {}
  render();
});

document.querySelector(".filters").addEventListener("click", e => {
  const button = e.target.closest("button[data-category]");
  if (!button) return;
  category = button.dataset.category;
  document.querySelectorAll(".filters button").forEach(b =>
    b.classList.toggle("active", b === button)
  );
  render();
});

search.addEventListener("input", render);
render();
</script>
